-- create bucket 'hive'
-- create directory 'datasets' 
-- create 2 folders 'csv' and 'parquet' and upload the resp. files

CREATE SCHEMA hive.hp
WITH (location = 's3a://hive/');


CREATE TABLE hive.hp.nyc_taxi_zone_hive (
    location_id VARCHAR,
    borough VARCHAR,
    zone VARCHAR,
    service_zone VARCHAR
)
WITH (
    external_location = 's3a://hive/datasets/csv',
    format = 'CSV',
    skip_header_line_count = 1
);

select count(*) from hive.hp.nyc_taxi_zone_hive ;

CREATE TABLE hive.hp.taxi_hive (
    VendorID              bigint,
    tpep_pickup_datetime  timestamp,
    tpep_dropoff_datetime timestamp,
    passenger_count       double,
    trip_distance         double,
    RatecodeID            double,
    store_and_fwd_flag    varchar,
    PULocationID          bigint,
    DOLocationID          bigint,
    payment_type          bigint,
    fare_amount           double,
    extra                 double,
    mta_tax               double,
    tip_amount            double,
    tolls_amount          double,
    improvement_surcharge double,
    total_amount          double,
    congestion_surcharge  double,
    airport_fee           double
)
WITH (
    external_location = 's3a://hive/datasets/parquet',
    format = 'PARQUET'
);


select count(*) from hive.hp.taxi_hive;

select count(*) from hive.hp.nyc_taxi_zone_hive;

select * from hive.hp.nyc_taxi_zone_hive limit 10;

select 
VendorID,
tpep_pickup_datetime,
tpep_dropoff_datetime,
fare_amount
from hive.hp.taxi_hive limit 10;
